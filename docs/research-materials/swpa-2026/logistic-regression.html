def log_analysis(file_path, title):
    print(f"\n{'='*20} {title.upper()} ANALYSIS {'='*20}")
    dataset = pd.read_csv(file_path)

    # Feature columns and target variable
    X = dataset[['mention_trust', 'mention_breach', 'mention_transparency', 
                'mention_risk', 'mention_privacy', 'mention_legal', 'mention_benefit']]
    Y = (dataset['sentiment_score'] <= -0.05).astype(int)

    X_stat = sm.add_constant(X)
    try:
        model = sm.Logit(Y, X_stat).fit(disp=0)
        
        # Creating a clean results table
        results = pd.DataFrame({
            'Odds Ratio': np.exp(model.params),
            'P-Value': model.pvalues
        })
        
        # Sort by Odds Ratio so the most important factors are at the top
        print(results.sort_values(by='Odds Ratio', ascending=False))
        
    except:
        print(f"Note: Not enough data in {title} to calculate statistics.")

    X_train, X_test, Y_train, Y_test = train_test_split(X, Y, test_size=0.25, random_state=0) #og was =16

    # Scale the numerical features
    scaler = StandardScaler()
    X_train = scaler.fit_transform(X_train)
    X_test = scaler.transform(X_test)

    # Model fit
    classifier = LogisticRegression(random_state=0, class_weight='balanced')
    classifier.fit(X_train, Y_train)
    Y_predict = classifier.predict(X_test)

    print(f"Confusion Matrix for {title}:")
    print(confusion_matrix(Y_test, Y_predict))

    print(f"\nClassification Report for {title}:")
    targets = ['Neutral/Positive', 'Negative']
    print(classification_report(Y_test, Y_predict, target_names=targets))

    plt.figure(figsize=(6, 4))
    sns.heatmap(confusion_matrix(Y_test, Y_predict), annot=True, fmt='d', cmap='Blues')
    plt.title(f'Confusion Matrix - {title}')
    plt.xlabel('Predicted')
    plt.ylabel('Actual')
    plt.show()

    return classifier
