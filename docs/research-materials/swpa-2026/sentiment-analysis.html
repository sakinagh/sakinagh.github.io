def process_file_sentiment(file_path):
    df = pd.read_csv(file_path)
    
    # Labeling of professionals
    def identify_role(row):
        text = (str(row.get('body', '')) + " " + str(row.get('text', ''))).lower()
        provider_markers = [
            'my practice', 'my patients', 'as a doctor', 'i am a gp', 'working in healthcare', 
            'my clinic', 'as a clinician', 'my patient'
        ]
        return 'Professional' if any(m in text for m in provider_markers) else 'General Public'

    # VADER sentiment and 7 feature clustering
    def get_context_and_sentiment(text):
        text = str(text).lower()
        v_score = sid.polarity_scores(text)['compound']
        
        clusters = {
            'mention_trust': ['trust', 'reliable', 'depend', 'honest', 'believe', 'confidence', 'consented'],
            'mention_breach': ['breach', 'leak', 'hacked', 'stolen', 'exposed', 'compromised'],
            'mention_transparency': ['transparent', 'clear', 'open', 'disclosure', 'explain', 'hidden', 'secret'],
            'mention_risk': ['risk', 'dangerous', 'hallucinate', 'wrong', 'mistake', 'error', 'scary', 'warning'],
            'mention_privacy': ['privacy', 'private', 'security', 'protected', 'hipaa', 'confidential'],
            'mention_legal': ['legal', 'lawsuit', 'compliant', 'standard', 'unethical', 'regulation'],
            'mention_benefit': ['helpful', 'efficient', 'save', 'time', 'improve', 'quality', 'automate', 'easy']
        }
        
        feats = {col: (1 if any(k in text for k in keywords) else 0) for col, keywords in clusters.items()}
        
        # Qualitative match tracking
        matches = []
        for col, kw in clusters.items():
            found = [k for k in kw if k in text]
            if found: matches.extend(found)
        
        return pd.Series({**feats, 'sentiment_score': v_score, 'matched_keywords': ", ".join(list(set(matches)))})

    df['identity_label'] = df.apply(identify_role, axis=1)
    
    # Feature extraction
    feature_results = df['body'].apply(get_context_and_sentiment)
    df = pd.concat([df, feature_results], axis=1)

    # Final Cleaning (Noise Removal)
    df = df[df['discussion_type'] != 'Technical/Infrastructure']
    df = df[~df['body'].str.contains("I am a bot|Please respect this", na=False, case=False)]
    df = df[df['body'] != '[No Comments]']
    
    final_cols = [
        'post_id', 'title', 'subreddit', 'comment_id', 'body', 
        'identity_label', 'matched_keywords', 'sentiment_score', 
        'mention_trust', 'mention_breach', 'mention_transparency', 
        'mention_risk', 'mention_privacy', 'mention_legal', 'mention_benefit'
    ]
    
    df[final_cols].to_csv('sentiment data/final_ml_all.csv', index=False)
    df[df['identity_label'] == 'Professional'][final_cols].to_csv('sentiment data/final_ml_professional.csv', index=False)
    df[df['identity_label'] == 'General Public'][final_cols].to_csv('sentiment data/final_ml_public.csv', index=False)
    
    print("All 3 files saved successfully in 'cleaned data/' folder.")
